cmake_minimum_required(VERSION 3.25)
project(FluidAmrexDependency LANGUAGES C CXX)

if(MSVC)
    set(OpenMP_RUNTIME_MSVC llvm)
    set(OpenMP_CXX_FLAGS "/openmp:llvm" CACHE STRING "" FORCE)
endif()
set(AMReX_SPACEDIM 2 CACHE STRING "" FORCE)
set(AMReX_PRECISION DOUBLE CACHE STRING "" FORCE)
set(AMReX_GPU_BACKEND NONE CACHE STRING "" FORCE)
set(AMReX_MPI OFF CACHE BOOL "" FORCE)
set(AMReX_OMP ON CACHE BOOL "" FORCE)
set(AMReX_FORTRAN OFF CACHE BOOL "" FORCE)
set(AMReX_EB OFF CACHE BOOL "" FORCE)
set(AMReX_PARTICLES OFF CACHE BOOL "" FORCE)
set(AMReX_AMRLEVEL OFF CACHE BOOL "" FORCE)
set(AMReX_LINEAR_SOLVERS_EM OFF CACHE BOOL "" FORCE)
set(AMReX_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(HYDRO_SPACEDIM 2 CACHE STRING "" FORCE)
set(HYDRO_PRECISION DOUBLE CACHE STRING "" FORCE)
set(HYDRO_GPU_BACKEND NONE CACHE STRING "" FORCE)
set(HYDRO_MPI OFF CACHE BOOL "" FORCE)
set(HYDRO_OMP ON CACHE BOOL "" FORCE)
set(HYDRO_EB OFF CACHE BOOL "" FORCE)
set(HYDRO_PROJECTIONS ON CACHE BOOL "" FORCE)
add_subdirectory("${FLUID_AMREX_SOURCE}" amrex)
add_subdirectory("${FLUID_HYDRO_SOURCE}" amrex-hydro)
