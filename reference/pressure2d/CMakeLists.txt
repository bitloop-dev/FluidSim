cmake_minimum_required(VERSION 3.25)
project(FluidPressure2D LANGUAGES CXX)

add_library(FluidPressure2D STATIC ../../FluidSim/AmrPressure2D.cpp ../../FluidSim/AmrPressureTopology2D.cpp)
target_include_directories(FluidPressure2D PUBLIC ../../FluidSim)
target_compile_features(FluidPressure2D PUBLIC cxx_std_20)
if(MSVC)
    set(OpenMP_RUNTIME_MSVC llvm)
    set(OpenMP_CXX_FLAGS "/openmp:llvm" CACHE STRING "" FORCE)
endif()
find_package(OpenMP QUIET)
if(OpenMP_CXX_FOUND)
    target_link_libraries(FluidPressure2D PUBLIC OpenMP::OpenMP_CXX)
endif()

option(PRESSURE2D_REFERENCE_TESTS "Compare with the pinned AMReX implementation" ON)
if(PRESSURE2D_REFERENCE_TESTS)
    find_package(AMReX CONFIG REQUIRED)
    enable_testing()
    add_executable(FluidPressure2DTests ../../tests/AmrPressure2DTests.cpp)
    target_link_libraries(FluidPressure2DTests PRIVATE FluidPressure2D AMReX::amrex)
    add_test(NAME Pressure2DReference COMMAND FluidPressure2DTests)
    set_tests_properties(Pressure2DReference PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=8;OMP_DYNAMIC=FALSE")
endif()
