cmake_minimum_required(VERSION 3.25)
project(FluidPressureGpuReference LANGUAGES CXX)

set(FLUID_REFERENCE_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../build/amrex-reference-deps" CACHE PATH "Pinned reference sources")
set(AMReX_SPACEDIM 2 CACHE STRING "" FORCE)
set(AMReX_GPU_BACKEND HIP CACHE STRING "Native GPU backend")
set(AMReX_PRECISION DOUBLE CACHE STRING "Reference scalar precision")
set(AMReX_MPI OFF CACHE BOOL "" FORCE)
set(AMReX_OMP OFF CACHE BOOL "" FORCE)
set(AMReX_FORTRAN OFF CACHE BOOL "" FORCE)
set(AMReX_EB OFF CACHE BOOL "" FORCE)
set(AMReX_PARTICLES OFF CACHE BOOL "" FORCE)
set(AMReX_AMRLEVEL OFF CACHE BOOL "" FORCE)
set(AMReX_LINEAR_SOLVERS_EM OFF CACHE BOOL "" FORCE)
set(AMReX_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${FLUID_REFERENCE_DEPS}/amrex" amrex)

add_executable(FluidPressureNativeReference native.cpp ../../FluidSim/AmrPressure2D.cpp ../../FluidSim/AmrPressureTopology2D.cpp)
target_include_directories(FluidPressureNativeReference PRIVATE ../../FluidSim)
target_compile_features(FluidPressureNativeReference PRIVATE cxx_std_20)
target_link_libraries(FluidPressureNativeReference PRIVATE amrex)
