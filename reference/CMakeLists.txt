cmake_minimum_required(VERSION 3.24)
project(FluidAmrexReference LANGUAGES C CXX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    set(OpenMP_RUNTIME_MSVC llvm)
    set(OpenMP_CXX_FLAGS "/openmp:llvm" CACHE STRING "" FORCE)
endif()

find_package(OpenMP REQUIRED)

# Kept independent of the scene build while the numerical contract is validated.
set(FLUID_REFERENCE_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../build/amrex-reference-deps" CACHE PATH "Pinned source checkouts")
set(AMReX_SPACEDIM 2 CACHE STRING "" FORCE)
set(AMReX_PRECISION DOUBLE CACHE STRING "" FORCE)
set(AMReX_GPU_BACKEND NONE CACHE STRING "" FORCE)
set(AMReX_MPI OFF CACHE BOOL "" FORCE)
set(AMReX_OMP ON CACHE BOOL "" FORCE)
set(AMReX_FORTRAN OFF CACHE BOOL "" FORCE)
set(AMReX_EB OFF CACHE BOOL "" FORCE)
set(AMReX_PARTICLES OFF CACHE BOOL "" FORCE)
set(AMReX_AMRLEVEL OFF CACHE BOOL "" FORCE)
set(AMReX_LINEAR_SOLVERS_EM OFF CACHE BOOL "" FORCE)
set(AMReX_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(HYDRO_SPACEDIM 2 CACHE STRING "" FORCE)
set(HYDRO_PRECISION DOUBLE CACHE STRING "" FORCE)
set(HYDRO_GPU_BACKEND NONE CACHE STRING "" FORCE)
set(HYDRO_MPI OFF CACHE BOOL "" FORCE)
set(HYDRO_OMP ON CACHE BOOL "" FORCE)
set(HYDRO_EB OFF CACHE BOOL "" FORCE)
set(HYDRO_PROJECTIONS ON CACHE BOOL "" FORCE)
add_subdirectory("${FLUID_REFERENCE_DEPS}/amrex" amrex)
add_subdirectory("${FLUID_REFERENCE_DEPS}/amrex-hydro" amrex-hydro)

add_executable(FluidAmrexReference mac_reference.cpp)
target_compile_features(FluidAmrexReference PRIVATE cxx_std_20)
target_link_libraries(FluidAmrexReference PRIVATE amrex_hydro)
add_library(FluidPressure2D ../FluidSim/AmrPressure2D.cpp ../FluidSim/AmrPressureTopology2D.cpp)
target_include_directories(FluidPressure2D PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../FluidSim")
target_compile_features(FluidPressure2D PUBLIC cxx_std_20)
target_link_libraries(FluidPressure2D PUBLIC OpenMP::OpenMP_CXX)
add_library(FluidAmrGas ../FluidSim/FluidGridAdaptive.cpp ../FluidSim/AmrField2D.cpp ../FluidSim/AmrTransfer2D.cpp ../FluidSim/AmrGasRuntime.cpp)
target_link_libraries(FluidAmrGas PUBLIC FluidPressure2D)
add_library(FluidUpstreamReference amrex/AmrGas.cpp amrex/AmrGasRuntime.cpp)
target_include_directories(FluidUpstreamReference PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../FluidSim")
target_compile_features(FluidUpstreamReference PUBLIC cxx_std_20)
target_link_libraries(FluidUpstreamReference PUBLIC amrex_hydro)
add_executable(FluidAmrGasTests amr_gas_tests.cpp)
target_link_libraries(FluidAmrGasTests PRIVATE FluidUpstreamReference)
add_executable(FluidExtractionTests extraction_tests.cpp)
target_link_libraries(FluidExtractionTests PRIVATE FluidAmrGas FluidUpstreamReference)
add_executable(FluidPressure2DTests ../tests/AmrPressure2DTests.cpp)
target_link_libraries(FluidPressure2DTests PRIVATE FluidPressure2D amrex)
enable_testing()
add_test(NAME composite_mac_reference COMMAND FluidAmrexReference n_cell=64 repeats=1)
add_test(NAME conservative_amr_gas_reference COMMAND FluidAmrGasTests mode=check n_cell=64)
add_test(NAME cpu_extraction_reference COMMAND FluidExtractionTests)
add_test(NAME pressure_extraction_reference COMMAND FluidPressure2DTests)
set_tests_properties(composite_mac_reference conservative_amr_gas_reference cpu_extraction_reference pressure_extraction_reference
    PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=8;OMP_DYNAMIC=FALSE")
